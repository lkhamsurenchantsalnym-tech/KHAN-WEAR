<script>
const items = [
    {id:1, name:"Цамц", price:35000},
    {id:2, name:"Hoodie", price:55000}
];

let cart = [];

function render(){
    let html = "";
    items.forEach(p=>{
        html += `
        <div class="card">
            <h3>${p.name}</h3>
            <p>${p.price}₮</p>
            <select id="size-${p.id}">
                <option>S</option>
                <option>M</option>
                <option>L</option>
            </select>
            <button onclick="add(${p.id})">Сагсанд нэмэх</button>
        </div>`;
    });
    document.getElementById("products").innerHTML = html;
}

function add(id){
    const size = document.getElementById("size-"+id).value;
    const product = items.find(p=>p.id===id);
    cart.push({...product, size});
    updateCart();
}

function updateCart(){
    let total = 0;
    let html = "";
    cart.forEach(i=>{
        total += i.price;
        html += `<li>${i.name} (${i.size}) - ${i.price}₮</li>`;
    });
    document.getElementById("cart").innerHTML = html;
    document.getElementById("total").innerText = total;
}

function order(){
    alert("Захиалга илгээгдлээ! (Demo)");
    cart = [];
    updateCart();
}

render();
</script>
